 46    The vibronic part of a defect is often captured by a simple harmonic oscillator.

128    @classmethod
129    def from_vaspruns(
130        cls,
131        vaspruns: list[Vasprun],
132        charge_state: int,
133        spin_index: int | None = None,
134        relaxed_index: int | None = None,
135        defect_band: Sequence[tuple] | None = None,
136        procar: Procar | None = None,
137        store_bandstructure: bool = False,
138        get_band_structure_kwargs: dict | None = None,
139        band_window: int = 7,
140        **kwargs,
141    ) -> HarmonicDefect:
142        """Create a HarmonicDefect from a list of vasprun.
143
144        .. note::
145            The constructor check that you have the vaspruns sorted by the distortions
146            but does not perform the sorting for you.  This serves as a safety check to
147            ensure that the provided vaspruns are properly ordered.  This check might
148            become optional in the future.
149
150        Args:
151            vaspruns: A list of Vasprun objects.
152            charge_state: The charge state for the defect.
153            spin_index: The index of the spin channel to use. If None, the spin channel
154                is determined by the channel with the most localized state.
155            relaxed_index: The index of the relaxed structure in the list of structures.
156            defect_band: The the index of the defect band since the defect for different
157                kpoints and spins presented as `[(band, kpt, spin), ...]`.
158            procar: A Procar object.  Used to identify the defect band if the defect_band_index is not provided.
159            store_bandstructure: Whether to store the bandstructure of the relaxed defect calculation.
160                Defaults to False to save space.
161            get_band_structure_kwargs: Keyword arguments to pass to the ``get_band_structure`` method.
162            band_window: The number of bands above and below the defect band to include when searching for the defect band.
163            **kwargs: Additional keyword arguments to pass to the constructor.
164
165        Returns:
166            A HarmonicDefect object.
167        """
168
169        def _parse_vasprun(vasprun: Vasprun) -> tuple[float, Structure]:
170            energy = vasprun.final_energy
171            struct = vasprun.final_structure
172            return (energy, struct)
173
174        energy_struct = list(map(_parse_vasprun, vaspruns))
175        unsorted_e = [e for e, _ in energy_struct]
176
177        if relaxed_index is None:
178            # Use the vasprun with the lowest energy
179            relaxed_index = np.argmin([e for e, _ in energy_struct])
180
181        sorted_list, distortions = sort_positive_definite(
182            energy_struct,
183            energy_struct[relaxed_index],
184            energy_struct[-1],
185            lambda x, y: get_dQ(x[1], y[1]),
186        )
187        energies, structures = list(zip(*sorted_list, strict=False))
188
189        if not np.allclose(unsorted_e, energies, atol=1e-99):  # pragma: no cover
190            msg = "The vaspruns should already be in order."
191            raise ValueError(msg)
192
193        omega = _get_omega(
194            Q=distortions,
195            E=energies,
196            Q0=distortions[relaxed_index],
197            E0=energies[relaxed_index],
198        )
199
200        get_band_structure_kwargs = get_band_structure_kwargs or {}
201        bandstructure = vaspruns[relaxed_index].get_band_structure(
202            **get_band_structure_kwargs,
203        )
204        ispin = vaspruns[relaxed_index].parameters["ISPIN"]
205
206        bs = bandstructure if store_bandstructure else None
207
208        if defect_band is None:
209            if procar is None:  # pragma: no cover
210                msg = "If defect_band_index is not provided, you must provide a Procar object."
211                raise ValueError(
212                    msg,
213                )
214            # Get the defect bands
215            defect_band_2s = list(
216                get_localized_states(
217                    bandstructure=bandstructure,
218                    procar=procar,
219                    band_window=band_window,
220                ),
221            )
222            defect_band_2s.sort(key=lambda x: (x[2], x[1]))
223            # group by the spin index
224            defect_band_grouped = {
225                spin: list(bands)
226                for spin, bands in groupby(defect_band_2s, lambda x: x[2])
227            }
228            min_localization = {
229                spin: np.min([val for _, _, _, val in bands])
230                for spin, bands in defect_band_grouped.items()
231            }
232            if spin_index is None:
233                # get the most localized spin
234                spin_index = min(min_localization, key=min_localization.get)
235            # drop the val
236            defect_band = [r_[:3] for r_ in defect_band_grouped[spin_index]]
237
238        return cls(
239            omega=omega,
240            charge_state=charge_state,
241            ispin=ispin,
242            structures=list(structures),
243            distortions=distortions,
244            energies=energies,
245            defect_band=defect_band,
246            relaxed_index=relaxed_index,
247            relaxed_bandstructure=bs,
248            vrun=vaspruns[relaxed_index],
249            **kwargs,
250        )

254        cls,
255        directories: list[Path],
256        charge_state: int | None = None,
257        relaxed_index: int | None = None,
258        defect_band: Sequence[tuple] | None = None,

260        get_band_structure_kwargs: dict | None = None,
261        **kwargs,
262    ) -> HarmonicDefect:
263        """Create a HarmonicDefect from a list of directories.
264
265        Args:
266            directories: A list of directories containing the vasprun.xml files.
267            charge_state: The charge state for the defect. If None, the charge state
268                will be determined using the data in the vasprun.xml and POTCAR files.
269            relaxed_index: The index of the relaxed structure in the list of structures.
270            defect_band: The the index of the defect band since the defect for different
271                kpoints and spins presented as `[(band, kpt, spin), ...]`.
272            store_bandstructure: Whether to store the bandstructure of the relaxed defect calculation.
273                Defaults to False to save space.
274            get_band_structure_kwargs: Keyword arguments to pass to the ``get_band_structure`` method.
275            **kwargs: Additional keyword arguments to pass to the constructor.
276
277        Returns:
278            A HarmonicDefect object.
279        """
280        vaspruns = [Vasprun(d / "vasprun.xml") for d in directories]
281        min_idx = np.argmin([v.final_energy for v in vaspruns])
282        min_dir = directories[min_idx]
283        procar_file = get_zfile(min_dir, "PROCAR")
284        procar = Procar(procar_file) if procar_file else None
285
286        if charge_state is None:
287            if vaspruns[min_idx].final_structure._charge is None:
288                msg = (
289                    "Charge state is not provided and cannot be parsed from the POTCAR."
290                )
291                raise ValueError(
292                    msg,
293                )
294            charge_state = vaspruns[0].final_structure.charge
295
296        if any(v.final_structure.charge != charge_state for v in vaspruns):
297            msg = "All vaspruns must have the same charge state."
298            raise ValueError(msg)
299
300        return cls.from_vaspruns(
301            vaspruns=vaspruns,
302            charge_state=charge_state,
303            relaxed_index=relaxed_index,
304            defect_band=defect_band,
305            procar=procar,
306            store_bandstructure=store_bandstructure,
307            get_band_structure_kwargs=get_band_structure_kwargs,
308            **kwargs,
309        )

354    def get_elph_me(self, defect_state: tuple) -> npt.ArrayLike:
355        """Calculate the electron phonon matrix elements.
356
357        Combine the data from the WSWQs to calculate the electron phonon matrix elements.
358        The matrix elements are calculated by combining the finite difference from the matrix overlaps.
359
360        (e_i - e_f) d(<W|S|W(Q)>) / dQ
361
362        And the eigenvalue difference.
363
364        Args:
365            defect_state: The defect state as a tuple of (band, kpoint, spin).
366                Note that even though the defect state is localized and does not have a well defined kpoint,
367                the band-edge states are usually dispersive and the k-point provided here should match the
368                k-point associated with the band-edge states.
369
370        Returns:
371            npt.ArrayLike: The electron phonon matrix elements from the defect band to all other bands.
372                The indices are [band_j,]
373        """
374        if self.wswqs is None:
375            msg = "WSWQs have not been read. Use `read_wswqs` first."
376            raise RuntimeError(msg)
377        distortions = [wswq["Q"] for wswq in self.wswqs]
378        wswqs = [wswq["wswq"] for wswq in self.wswqs]
379        band_index, kpoint_index, spin_index = defect_state
380        # The second band index is associated with the defect state
381        # since we are usually interested in capture
382        slopes = _get_wswq_slope(distortions, wswqs)[
383            spin_index,
384            kpoint_index,
385            :,
386            band_index,
387        ]
388        ediffs = self._get_ediff(output_order="skb")[spin_index, kpoint_index, :]
389        return np.multiply(slopes, ediffs)

391    def _get_ediff(self, output_order: str = "skb") -> npt.NDArray:
392        """Compute the eigenvalue difference to the defect band.
393
394        .. note::
395            Since the different matrix element output files have different index orders,
396            But most calculations require the energies, we should always perform the
397            rearrangement here so that we have a single point of failure.
398
399        Args:
400            output_order: The order of the output. Defaults to "skb" (spin, kpoint, band]).
401                You can also use "bks" (band, kpoint, spin).
402
403        Returns:
404            The eigenvalue difference to the defect band in the order specified by output_order.
405
406        """
407        if self.relaxed_bandstructure is None:  # pragma: no cover
408            msg = (
409                "The ``relaxed_bandstructure`` must be set before ``ediff`` can be computed. "
410                "Try setting ``store_bandstructure=True`` when initializing."
411            )
412            raise ValueError(
413                msg,
414            )
415
416        ediffs_ = _get_ks_ediff(
417            bandstructure=self.relaxed_bandstructure,
418            defect_band=self.defect_band,
419        )
420        ediffs_stack = [
421            ediffs_[Spin.up].T,
422        ]
423        if Spin.down in ediffs_:
424            ediffs_stack.append(ediffs_[Spin.down].T)
425        ediffs = np.stack(ediffs_stack)
426
427        if output_order == "skb":
428            return ediffs
429        if output_order == "bks":
430            return ediffs.transpose((2, 1, 0))
431        msg = "Invalid output_order, choose from 'skb' or 'bks'."
432        raise ValueError(
433            msg,
434        )

436    def get_dielectric_function(
437        self,
438        idir: int,
439        jdir: int,
440    ) -> tuple[npt.ArrayLike, npt.ArrayLike, npt.ArrayLike]:
441        """Calculate the dielectric function.
442
443        Args:
444            idir: The first direction of the dielectric tensor.
445            jdir: The second direction of the dielectric tensor.
446
447        Returns:
448            energy: The energy grid representing the dielectric function.
449            eps_vbm: The dielectric function from the VBM to the defect state.
450            eps_cbm: The dielectric function from the defect state to the CBM.
451        """
452        dfc = DielectricFunctionCalculator.from_vasp_objects(
453            vrun=self.vrun,
454            waveder=self.waveder,
455        )
456
457        # two masks to select for VBM -> Defect and Defect -> CBM
458        mask_vbm = np.zeros_like(dfc.cder_real)
459        mask_cbm = np.zeros_like(dfc.cder_real)
460        min_def_eig, max_def_eig = np.inf, -np.inf
461        for ib, ik, ispin in self.defect_band:
462            mask_vbm[:ib, ib, ik, ispin] = 1.0
463            mask_cbm[ib, ib:, ik, ispin] = 1.0
464            min_def_eig = min(dfc.eigs[ib, ik, ispin], min_def_eig)
465            max_def_eig = max(dfc.eigs[ib, ik, ispin], max_def_eig)
466
467        # VBM must be lower than defect and CBM must be higher than defect
468        # For situations where there are multiple defect states, hopefully
469        # the Fermi smearing will reduce the transition between defect states
470        # and we will only measure transitions to the band edges.
471        e_vbm = np.max(dfc.eigs[dfc.eigs < min_def_eig])
472        e_cbm = np.min(dfc.eigs[dfc.eigs > max_def_eig])
473
474        fermi_vbm = (e_vbm + min_def_eig) / 2
475        fermi_cbm = (e_cbm + max_def_eig) / 2
476
477        energy, eps_vbm = dfc.get_epsilon(idir, jdir, fermi_vbm, mask=mask_vbm)
478        _, eps_cbm = dfc.get_epsilon(idir, jdir, fermi_cbm, mask=mask_cbm)
479
480        return energy, eps_vbm, eps_cbm

505def get_SRH_coefficient(
506    initial_state: HarmonicDefect,
507    final_state: HarmonicDefect,
508    defect_state: tuple[int, int, int],
509    T: float | npt.ArrayLike,
510    dE: float,
511    g: int = 1,
512    occ_tol: float = 1e-3,
513    n_band_edge: int = 1,
514    use_final_state_elph: bool = False,
515) -> npt.ArrayLike:
516    """Get the SRH coefficient for a defect.
517
518    Args:
519        initial_state: The initial charge state of the defect.
520        final_state: The final charge state of the defect.
521        defect_state: The band, k-point, and spin of the defect.
522        T: The temperature in Kelvin.
523        dE: The energy difference between the defect and the band edge.
524        g: The degeneracy of the defect state.
525        occ_tol: The tolerance for determining if a state is occupied.
526        n_band_edge: The number of bands to average over at the band edge.
527        use_final_state_elph: Whether to use the final state's ELPH data.
528            This is useful if the initial state does not have a well-defined
529            defect state.
530
531    Returns:
532        The SRH recombination coefficient in units of cm^3 s^-1.
533    """
534    if use_final_state_elph:
535        me_all = final_state.get_elph_me(defect_state=defect_state)
536    else:
537        me_all = initial_state.get_elph_me(defect_state=defect_state)
538    defect_band, _, _ = defect_state
539
540    if initial_state.charge_state == final_state.charge_state + 1:
541        band_slice = slice(defect_band + 1, defect_band + 1 + n_band_edge)
542    elif initial_state.charge_state == final_state.charge_state - 1:
543        band_slice = slice(defect_band - n_band_edge, defect_band)
544    else:
545        msg = "SRH capture event must involve a charge state change of 1."
546        raise ValueError(msg)
547
548    me_band_edge = me_all[band_slice]
549    dQ = get_dQ(initial_state.relaxed_structure, final_state.relaxed_structure)
550    volume = initial_state.relaxed_structure.volume
551    return get_SRH_coef(
552        T,
553        dQ=dQ,
554        dE=dE,
555        omega_i=initial_state.omega_eV,
556        omega_f=final_state.omega_eV,
557        elph_me=np.average(me_band_edge),
558        volume=volume,
559        g=g,
560        occ_tol=occ_tol,
561    )

564def get_dQ(ground: Structure, excited: Structure) -> float:
565    """Calculate configuration coordinate difference.
566
567    Args:
568        ground : pymatgen structure corresponding to the ground (final) state
569        excited : pymatgen structure corresponding to the excited (initial) state
570
571    Returns:
572        (float):  the dQ value (amu^{1/2} Angstrom)
573    """
574    return np.sqrt(
575        np.sum(
576            [
577                x[0].distance(x[1]) ** 2 * x[0].specie.atomic_mass
578                for x in zip(ground, excited, strict=False)
579            ],
580        ),
581    )

610    Q0: float,

616        """Get the parabola function."""
617        return 0.5 * omega**2 * (Q - Q0) ** 2 + E0
618

623def _get_wswq_slope(distortions: list[float], wswqs: list[WSWQ]) -> npt.NDArray:
624    """Get the slopes of the overlap matrixs vs. Q.
625
626    Args:
627        distortions: List of Q values (amu^{1/2} Angstrom).
628        wswqs: List of WSWQ objects.
629
630    Returns:
631        npt.NDArray: slope matrix with the same shape as the ``WSWQ.data``.
632            Since there is always ambiguity in the phase, we require that the output
633            is always positive.
634    """
635    yy = np.stack(
636        [
637            np.abs(ww.data) * np.sign(qq)
638            for qq, ww in zip(distortions, wswqs, strict=False)
639        ]
640    )
641    _, *oldshape = yy.shape
642    return np.polyfit(distortions, yy.reshape(yy.shape[0], -1), deg=1)[0].reshape(
643        *oldshape,
644    )

647def _get_ks_ediff(
648    bandstructure: BandStructure,
649    defect_band: Sequence[tuple],
650) -> dict[Spin, npt.NDArray]:
651    """Calculate the Kohn-Sham energy between the defect state.
652
653    Get the eigenvalue differences to the defect band. Report this difference
654    on each k-point and each spin, the result should be shape [nspins, nkpoints, nbands].
655
656    Args:
657        bandstructure: A BandStructure object.
658        defect_band: The defect band given as a list of tuples (band_index, kpoint_index, spin_index).
659
660    Returns:
661        npt.NDArray: The Kohn-Sham energy difference between the defect state and other states.
662        Indexed the same way as ``bandstructure.bands``.
663    """
664    res = {}
665    b_at_kpt_and_spin = {(k, s): b for b, k, s in defect_band}
666    for ispin, eigs in bandstructure.bands.items():
667        spin_index = 0 if ispin == Spin.up else 1
668        res[ispin] = np.zeros_like(eigs)
669        for ikpt, _kpt in enumerate(bandstructure.kpoints):
670            iband = b_at_kpt_and_spin.get((ikpt, spin_index))
671            if iband is None:
672                continue
673            e_at_def_band = eigs[iband, ikpt]
674            e_diff = eigs[:, ikpt] - e_at_def_band
675            res[ispin][:, ikpt] = e_diff
676    return res

357
358        Combine the data from the WSWQs to calculate the electron phonon matrix elements.
359        The matrix elements are calculated by combining the finite difference from the matrix overlaps.
360
361        (e_i - e_f) d(<W|S|W(Q)>) / dQ
362
363        And the eigenvalue difference.
364
365        Args:
366            defect_state: The defect state as a tuple of (band, kpoint, spin).
367                Note that even though the defect state is localized and does not have a well defined kpoint,
368                the band-edge states are usually dispersive and the k-point provided here should match the
369                k-point associated with the band-edge states.
370
371        Returns:
372            npt.ArrayLike: The electron phonon matrix elements from the defect band to all other bands.
373                The indices are [band_j,]
374        """
375        if self.wswqs is None:
376            msg = "WSWQs have not been read. Use `read_wswqs` first."
377            raise RuntimeError(msg)
378        distortions = [wswq["Q"] for wswq in self.wswqs]
379        wswqs = [wswq["wswq"] for wswq in self.wswqs]
380        band_index, kpoint_index, spin_index = defect_state
381        # The second band index is associated with the defect state
382        # since we are usually interested in capture
383        slopes = _get_wswq_slope(distortions, wswqs)[
384            spin_index,
385            kpoint_index,
386            :,
387            band_index,
388        ]
389        ediffs = self._get_ediff(output_order="skb")[spin_index, kpoint_index, :]
390        return np.multiply(slopes, ediffs)

392    def _get_ediff(self, output_order: str = "skb") -> npt.NDArray:
393        """Compute the eigenvalue difference to the defect band.
394
395        .. note::
396            Since the different matrix element output files have different index orders,
397            But most calculations require the energies, we should always perform the
398            rearrangement here so that we have a single point of failure.
399
400        Args:
401            output_order: The order of the output. Defaults to "skb" (spin, kpoint, band]).
402                You can also use "bks" (band, kpoint, spin).
403
404        Returns:
405            The eigenvalue difference to the defect band in the order specified by output_order.
406
407        """
408        if self.relaxed_bandstructure is None:  # pragma: no cover
409            msg = (
410                "The ``relaxed_bandstructure`` must be set before ``ediff`` can be computed. "
411                "Try setting ``store_bandstructure=True`` when initializing."
412            )
413            raise ValueError(
414                msg,
415            )
416
417        ediffs_ = _get_ks_ediff(
418            bandstructure=self.relaxed_bandstructure,
419            defect_band=self.defect_band,
420        )
421        ediffs_stack = [
422            ediffs_[Spin.up].T,
423        ]
424        if Spin.down in ediffs_:
425            ediffs_stack.append(ediffs_[Spin.down].T)
426        ediffs = np.stack(ediffs_stack)
427
428        if output_order == "skb":
429            return ediffs
430        if output_order == "bks":
431            return ediffs.transpose((2, 1, 0))
432        msg = "Invalid output_order, choose from 'skb' or 'bks'."
433        raise ValueError(
434            msg,
435        )

437    def get_dielectric_function(
438        self,
439        idir: int,
440        jdir: int,
441    ) -> tuple[npt.ArrayLike, npt.ArrayLike, npt.ArrayLike]:
442        """Calculate the dielectric function.
443
444        Args:
445            idir: The first direction of the dielectric tensor.
446            jdir: The second direction of the dielectric tensor.
447
448        Returns:
449            energy: The energy grid representing the dielectric function.
450            eps_vbm: The dielectric function from the VBM to the defect state.
451            eps_cbm: The dielectric function from the defect state to the CBM.
452        """
453        dfc = DielectricFunctionCalculator.from_vasp_objects(
454            vrun=self.vrun,
455            waveder=self.waveder,
456        )
457
458        # two masks to select for VBM -> Defect and Defect -> CBM
459        mask_vbm = np.zeros_like(dfc.cder_real)
460        mask_cbm = np.zeros_like(dfc.cder_real)
461        min_def_eig, max_def_eig = np.inf, -np.inf
462        for ib, ik, ispin in self.defect_band:
463            mask_vbm[:ib, ib, ik, ispin] = 1.0
464            mask_cbm[ib, ib:, ik, ispin] = 1.0
465            min_def_eig = min(dfc.eigs[ib, ik, ispin], min_def_eig)
466            max_def_eig = max(dfc.eigs[ib, ik, ispin], max_def_eig)
467
468        # VBM must be lower than defect and CBM must be higher than defect
469        # For situations where there are multiple defect states, hopefully
470        # the Fermi smearing will reduce the transition between defect states
471        # and we will only measure transitions to the band edges.
472        e_vbm = np.max(dfc.eigs[dfc.eigs < min_def_eig])
473        e_cbm = np.min(dfc.eigs[dfc.eigs > max_def_eig])
474
475        fermi_vbm = (e_vbm + min_def_eig) / 2
476        fermi_cbm = (e_cbm + max_def_eig) / 2
477
478        energy, eps_vbm = dfc.get_epsilon(idir, jdir, fermi_vbm, mask=mask_vbm)
479        _, eps_cbm = dfc.get_epsilon(idir, jdir, fermi_cbm, mask=mask_cbm)
480
481        return energy, eps_vbm, eps_cbm
