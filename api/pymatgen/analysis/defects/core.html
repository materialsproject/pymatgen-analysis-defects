 150    def get_charge_states(self, padding: int = 1) -> list[int]:
 151        """Potential charge states for a given oxidation state.
 152
 153        If user charges are specified, these will be returned.
 154        Otherwise, the charge states will be determined automatically based
 155        on the oxidation state with a padding on either sites of 0 and the
 156        oxidation state value.
 157
 158        Args:
 159            padding: The number of charge states on the on either side of
 160                0 and the oxidation state.
 161
 162        Returns:
 163            list of possible charge states
 164        """
 165        if self.user_charges:
 166            return self.user_charges
 167
 168        if isinstance(self.oxi_state, int) or self.oxi_state.is_integer():
 169            oxi_state = int(self.oxi_state)
 170        else:  # pragma: no cover
 171            sign = -1 if self.oxi_state < 0 else 1
 172            oxi_state = sign * int(np.ceil(abs(self.oxi_state)))
 173            _logger.warning(
 174                "Non-integer oxidation state detected."
 175                "Round to integer with larger absolute value: %s -> %s",
 176                self.oxi_state,
 177                oxi_state,
 178            )
 179
 180        if oxi_state >= 0:
 181            charges = [*range(-padding, oxi_state + padding + 1)]
 182        else:
 183            charges = [*range(oxi_state - padding, padding + 1)]
 184
 185        return charges

 187    def get_supercell_structure(
 188        self,
 189        sc_mat: np.ndarray | None = None,
 190        defect_structure: Structure | None = None,
 191        dummy_species: str | None = None,
 192        min_atoms: int = 80,
 193        max_atoms: int = 240,
 194        min_length: float = 10.0,
 195        force_diagonal: bool = False,
 196        relax_radius: float | str | None = None,
 197        perturb: float | None = None,
 198        target_frac_coords: np.ndarray | None = None,
 199        return_site: bool = False,
 200    ) -> Structure:
 201        """Generate the supercell for a defect.
 202
 203        If the bulk structure (provided by `Defect.structure`) and defect structures
 204        have oxidation state information, then the supercell will be decorated with
 205        oxidation states.  Otherwise, the supercell structure will not have any oxidation
 206        state information.  This also allows for oxidation state decoration of different
 207        defect charge states.
 208
 209        .. code-block:: python
 210            defect_struct = defect.defect_structure.copy()
 211            defect_struct.add_oxidation_state_by_guess(target_charge=2)
 212
 213        .. note::
 214            Since any algorithm for decorating the oxidation states will be combinatorial,
 215            They can be very slow for large supercells. If you want to decorate the structure
 216            with oxidation states, you have to first decorate the smaller unit cell defect
 217            structure then implant it in the supercell.
 218
 219        Args:
 220            sc_mat: supercell matrix if None, the supercell will be determined by `CubicSupercellAnalyzer`.
 221            defect_structure: Alternative defect structure to use for generating the supercell.
 222                You might want to use this if you want to decorate the oxidation states of the
 223                defect structure using a custom method.
 224            dummy_species: Dummy species to highlight the defect position (for visualizing vacancies).
 225            max_atoms: Maximum number of atoms allowed in the supercell.
 226            min_atoms: Minimum number of atoms allowed in the supercell.
 227            min_length: Minimum length of the smallest supercell lattice vector.
 228            force_diagonal: If True, return a transformation with a diagonal transformation matrix.
 229            relax_radius: Relax the supercell atoms to a sphere of this radius around the defect site.
 230            perturb: The amount to perturb the sites in the supercell. Only perturb the sites with
 231                selective dynamics set to True. So this setting only works with `relax_radius`.
 232            target_frac_coords: If set, defect will be placed at the closest equivalent site to these
 233                fractional coordinates.
 234            return_site: If True, returns a tuple of the (defect supercell, defect site position).
 235
 236        Returns:
 237            Structure: The supercell structure.
 238            PeriodicSite (optional): The position of the defect site in the supercell.
 239        """
 240
 241        def _has_oxi(struct: Structure) -> bool:
 242            return all(hasattr(site.specie, "oxi_state") for site in struct)
 243
 244        if defect_structure is None:
 245            defect_structure = self.centered_defect_structure
 246        bulk_structure = center_structure(self.structure, self.site.frac_coords)
 247        keep_oxi = _has_oxi(bulk_structure) and _has_oxi(defect_structure)
 248
 249        if sc_mat is None:
 250            sc_mat = get_sc_fromstruct(
 251                bulk_structure,
 252                min_atoms=min_atoms,
 253                max_atoms=max_atoms,
 254                min_length=min_length,
 255                force_diagonal=force_diagonal,
 256            )
 257
 258        # Get the translation vector in to the desired cell in Cartesian coordinates
 259        if target_frac_coords is None:
 260            target_frac_coords = [1e-6, 1e-6, 1e-6]
 261        trans_R = np.floor(np.dot(target_frac_coords, sc_mat))
 262        trans_vec = np.dot(trans_R, bulk_structure.lattice.matrix)
 263        sc_defect_struct = bulk_structure * sc_mat
 264
 265        # Use the site as a reference point for the position
 266        sc_mat_inv = np.linalg.inv(sc_mat)
 267        sc_pos = np.dot(self.site.frac_coords, sc_mat_inv)
 268        sc_site = PeriodicSite(
 269            species=self.site.specie,
 270            coords=sc_pos,
 271            lattice=sc_defect_struct.lattice,
 272            coords_are_cartesian=False,
 273        )
 274
 275        bulk_site_mapping = _get_mapped_sites(bulk_structure, sc_defect_struct)
 276        defect_site_mapping = _get_mapped_sites(defect_structure, sc_defect_struct)
 277
 278        # Remove the indices that that are not mapped
 279        rm_indices = set(bulk_site_mapping.values()) - set(defect_site_mapping.values())
 280
 281        # Set the species for the sites that are mapped
 282        for defect_site, bulk_site in defect_site_mapping.items():
 283            sc_defect_struct[bulk_site]._species = defect_structure[defect_site].species
 284
 285        # interstitials
 286        int_uc_indices = set(range(len(defect_structure))) - set(
 287            defect_site_mapping.keys(),
 288        )
 289        for i in int_uc_indices:
 290            int_sc_pos = np.dot(defect_structure[i].frac_coords, sc_mat_inv)
 291            sc_defect_struct.insert(
 292                0,
 293                defect_structure[i].specie,
 294                int_sc_pos,
 295                coords_are_cartesian=False,
 296            )
 297
 298        # Remove the sites that are not mapped
 299        sc_defect_struct.remove_sites(list(rm_indices))
 300        _set_selective_dynamics(
 301            structure=sc_defect_struct,
 302            site_pos=sc_site.coords,
 303            relax_radius=relax_radius,
 304        )
 305        if perturb is not None:
 306            _perturb_dynamic_sites(sc_defect_struct, distance=perturb)
 307
 308        # Translate the structure to the target position
 309        sc_defect_struct.translate_sites(
 310            indices=list(range(len(sc_defect_struct))),
 311            vector=trans_vec,
 312            frac_coords=False,
 313        )
 314        sc_site._frac_coords += trans_R
 315
 316        if not keep_oxi:
 317            sc_defect_struct.remove_oxidation_states()
 318
 319        if dummy_species is not None:
 320            sc_defect_struct.insert(0, dummy_species, sc_site.frac_coords)
 321
 322        if return_site:
 323            return sc_defect_struct, sc_site
 324        return sc_defect_struct

 364class NamedDefect(MSONable):
 365    """Class for defect definition without the UC structure.
 366
 367    The defect is defined only by its name. For complexes the name
 368    should be created with "+" between the individual defects.
 369
 370    .. note::
 371        This class is only used to help aggregate defects calculated
 372        outside of our framework so a ``Defect`` object is missing.
 373        The object will not have any mechanism for generating supercells
 374        or guessing oxidation states.  It is simply a placeholder to help
 375        with the grouping logic of the Formation Energy diagram analysis.
 376
 377    """
 378
 379    def __init__(self, name: str, bulk_formula: str, element_changes: dict) -> None:
 380        """Initialize a NamedDefect object.
 381
 382        Args:
 383            name: The name of the defect.
 384            bulk_formula: The formula of the bulk structure.
 385            element_changes: The species changes of the defect.
 386        """
 387        self.name = name
 388        self.bulk_formula = bulk_formula
 389        self.element_changes = element_changes

 391    @classmethod
 392    def from_structures(
 393        cls, defect_structure: Structure, bulk_structure: Structure
 394    ) -> Self:
 395        """Initialize a NameDefect object from structures.
 396
 397        Args:
 398            defect_structure: The structure of the defect.
 399            bulk_structure: The structure of the bulk.
 400
 401        Returns:
 402            NamedDefect: The defect object.
 403        """
 404        el_diff = _get_el_changes_from_structures(defect_structure, bulk_structure)
 405        name_ = _get_defect_name(el_diff)
 406        bulk_formula = bulk_structure.composition.reduced_formula
 407        return cls(name=name_, bulk_formula=bulk_formula, element_changes=el_diff)

 411        """Get the latex name of the defect.
 412
 413        Returns:

 417        l_names = []
 418        for n in single_names:
 419            root, suffix = n.split("_")
 420            l_names.append(rf"{root}$_{{\rm {suffix}}}$")

 608    def _guess_oxi_state(self) -> float:
 609        """Best guess for the oxidation state of the defect.
 610
 611        For a substitution defect, the oxidation state of the defect is given
 612        by the difference between the oxidation state of the new and old atoms.
 613
 614        Returns:
 615            float: The oxidation state of the defect.
 616        """
 617        rm_oxi = self.structure[self.defect_site_index].specie.oxi_state
 618
 619        # check if substitution atom is present in structure (i.e. antisite substitution):
 620        sub_elt_sites_in_struct = [
 621            site
 622            for site in self.structure
 623            if site.specie.symbol
 624            == self.site.specie.symbol  # gives Element symbol (without oxi state)
 625        ]
 626        if len(sub_elt_sites_in_struct) == 0:
