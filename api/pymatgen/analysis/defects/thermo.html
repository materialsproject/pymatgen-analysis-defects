  99    def get_freysoldt_correction(
 100        self,
 101        defect_locpot: Locpot | dict,
 102        bulk_locpot: Locpot | dict,
 103        dielectric: float | NDArray,
 104        defect_struct: Structure | None = None,
 105        bulk_struct: Structure | None = None,
 106        **kwargs,
 107    ) -> CorrectionResult:
 108        """Calculate the Freysoldt correction.
 109
 110        Updates the corrections dictionary with the Freysoldt correction
 111        and returns the planar averaged potential data for plotting.
 112
 113        Args:
 114            defect_locpot:
 115                The Locpot object for the defect supercell.
 116                Or a dictionary of the planar averaged locpot
 117            bulk_locpot:
 118                The Locpot object for the bulk supercell.
 119                Or a dictionary of the planar averaged locpot
 120            dielectric:
 121                The dielectric tensor or constant for the bulk material.
 122            defect_struct:
 123                The defect structure. If None, the structure of the defect_locpot
 124                will be used.
 125            bulk_struct:
 126                The bulk structure. If None, the structure of the bulk_locpot
 127                will be used.
 128            **kwargs:
 129                Additional keyword arguments for the get_correction method.
 130
 131        Returns:
 132            dict:
 133                The plotting data to analyze the planar averaged electrostatic potential
 134                in the three periodic lattice directions.
 135        """
 136        if defect_struct is None:
 137            defect_struct = getattr(defect_locpot, "structure", None)
 138        if bulk_struct is None:
 139            bulk_struct = getattr(bulk_locpot, "structure", None)
 140
 141        if defect_struct is None or bulk_struct is None:  # pragma: no cover
 142            msg = "defect_struct and/or bulk_struct is missing either provide the structure or provide the complete locpot."
 143            raise ValueError(
 144                msg,
 145            )
 146
 147        if self.sc_defect_frac_coords is None:
 148            finder = DefectSiteFinder()
 149            defect_fpos = finder.get_defect_fpos(
 150                defect_structure=defect_struct,
 151                base_structure=bulk_struct,
 152            )
 153            self.sc_defect_frac_coords = defect_fpos
 154        else:  # pragma: no cover
 155            defect_fpos = self.sc_defect_frac_coords
 156
 157        if isinstance(defect_locpot, VolumetricData):
 158            defect_gn = defect_locpot.dim
 159        elif isinstance(defect_locpot, dict):
 160            defect_gn = tuple(map(len, (defect_locpot for k in ["0", "1", "2"])))
 161
 162        if isinstance(bulk_locpot, VolumetricData):
 163            bulk_sc_locpot = get_sc_locpot(
 164                uc_locpot=bulk_locpot,
 165                defect_struct=defect_struct,
 166                grid_out=defect_gn,
 167                up_sample=2,
 168            )
 169            bulk_locpot = bulk_sc_locpot
 170
 171        frey_corr = get_freysoldt_correction(
 172            q=self.charge_state,
 173            dielectric=dielectric,
 174            defect_locpot=defect_locpot,
 175            bulk_locpot=bulk_sc_locpot,
 176            defect_frac_coords=defect_fpos,
 177            lattice=defect_struct.lattice,
 178            **kwargs,
 179        )
 180        self.corrections.update(
 181            {
 182                "freysoldt": frey_corr.correction_energy,
 183            },
 184        )
 185        self.corrections_metadata.update({"freysoldt": frey_corr.metadata.copy()})
 186        return frey_corr

 225class FormationEnergyDiagram(MSONable):
 226    """Formation energy.
 227
 228    Attributes:
 229        defect_entries:
 230            The list of defect entries for the different charge states.
 231            The finite-size correction should already be applied to these.
 232        pd_entries:
 233            The list of entries used to construct the phase diagram and chemical
 234            potential diagram. They will be used to determine the stability region
 235            of the bulk crystal.  The entries are used instead of the ``PhaseDiagram``
 236            object to make serializing the object easier.
 237        vbm:
 238            The VBM of the bulk crystal.
 239        band_gap:
 240            The band gap of the bulk crystal.
 241        bulk_entry:
 242            The bulk computed entry to get the total energy of the bulk supercell.
 243            This is only used in case where the bulk entry data is not provided by
 244            the individual defect entries themselves. Default is None.
 245            The data from the `bulk_entry` attached to each `defect_entry` will be
 246            preferentially used if it is available.
 247        inc_inf_values:
 248            If False these boundary points at infinity are ignored when we look at the
 249            chemical potential limits.
 250            The stability region is sometimes unbounded, example:
 251            Mn_Ga in GaN, the chemical potential of Mn is does not affect
 252            the stability of GaN so it can go to ``-inf``.
 253            A artificial value is needed to help the half-space intersection algorithm.
 254            This can be justified since these tend to be the substitutional elements
 255            which should not have very negative chemical potential.
 256        bulk_stability:
 257            If the bulk energy is above the convex hull, lower it to this value below
 258            the convex hull.
 259    """
 260
 261    defect_entries: list[DefectEntry]
 262    pd_entries: list[ComputedEntry]
 263    vbm: float
 264    band_gap: float | None = None
 265    bulk_entry: ComputedStructureEntry | None = None
 266    inc_inf_values: bool = False
 267    bulk_stability: float = 0.001
 268
 269    def __post_init__(self) -> None:
 270        """Post-initialization.
 271
 272        - Reconstruct the phase diagram with the bulk entry
 273        - Make sure that the bulk entry is stable
 274        - create the chemical potential diagram using only the formation energies
 275        """
 276        g = group_defect_entries(self.defect_entries)
 277        if next(g, True) and next(g, False):
 278            msg = (
 279                "Defects are not of same type! "
 280                "Use MultiFormationEnergyDiagram for multiple defect types"
 281            )
 282            raise ValueError(
 283                msg,
 284            )
 285        # if all of the `DefectEntry` objects have the same `bulk_entry` then `self.bulk_entry` is not needed
 286        if self.bulk_entry is None and any(
 287            x.bulk_entry is None for x in self.defect_entries
 288        ):
 289            msg = "The bulk entry must be provided."
 290            raise RuntimeError(
 291                msg,
 292            )
 293
 294        bulk_entry = self.bulk_entry or min(
 295            [x.bulk_entry for x in self.defect_entries],
 296            key=lambda x: x.energy_per_atom,
 297        )
 298        pd_ = PhaseDiagram(self.pd_entries)
 299        entries = pd_.stable_entries | {bulk_entry}
 300        pd_ = PhaseDiagram(entries)
 301        self.phase_diagram = ensure_stable_bulk(pd_, bulk_entry, self.bulk_stability)
 302        entries = []
 303        for entry in self.phase_diagram.stable_entries:
 304            d_ = {
 305                "energy": self.phase_diagram.get_form_energy(entry),
 306                "composition": entry.composition,
 307                "entry_id": entry.entry_id,
 308                "correction": 0.0,
 309            }
 310            entries.append(ComputedEntry.from_dict(d_))
 311            entries.append(ComputedEntry.from_dict(d_))
 312        self.chempot_diagram = ChemicalPotentialDiagram(entries)
 313        if (
 314            bulk_entry.composition.reduced_formula not in self.chempot_diagram.domains
 315        ):  # pragma: no cover
 316            msg = (
 317                "Bulk entry is not stable in the chemical potential diagram."
 318                "Consider increasing the `bulk_stability` to make it more stable."
 319            )
 320            raise ValueError(
 321                msg,
 322            )
 323        chempot_limits = self.chempot_diagram.domains[
 324            bulk_entry.composition.reduced_formula
 325        ]
 326
 327        if self.inc_inf_values:
 328            self._chempot_limits_arr = chempot_limits
 329        else:
 330            boundary_value = self.chempot_diagram.default_min_limit
 331            self._chempot_limits_arr = chempot_limits[
 332                ~np.any(chempot_limits == boundary_value, axis=1)
 333            ]
 334
 335        self.dft_energies = {
 336            el: self.phase_diagram.get_hull_energy_per_atom(Composition(str(el)))
 337            for el in self.phase_diagram.elements
 338        }

 403    @classmethod
 404    def with_directories(
 405        cls,
 406        directory_map: dict[str, str | Path],
 407        defect: Defect,
 408        pd_entries: list[ComputedEntry],
 409        dielectric: float | NDArray,
 410        vbm: float | None = None,
 411        **kwargs,
 412    ) -> FormationEnergyDiagram:
 413        """Create a FormationEnergyDiagram from VASP directories.
 414
 415        Args:
 416            directory_map: A dictionary mapping the defect name to the directory containing the
 417                VASP calculation.
 418            defect: The defect used to create the defect entries.
 419            pd_entries: The list of entries used to construct the phase diagram and chemical
 420                potential diagram. They will be used to determine the stability region
 421                of the bulk crystal.
 422            dielectric: The dielectric constant of the bulk crystal.
 423            vbm: The VBM of the bulk crystal.
 424            **kwargs: Additional keyword arguments for the constructor.
 425        """
 426
 427        def _read_dir(directory: str | Path) -> tuple[ComputedEntry, Locpot]:
 428            directory = Path(directory)
 429            vr = Vasprun(get_zfile(Path(directory), "vasprun.xml"))
 430            ent = vr.get_computed_entry()
 431            locpot = Locpot.from_file(get_zfile(directory, "LOCPOT"))
 432            return ent, locpot
 433
 434        if "bulk" not in directory_map:
 435            msg = "The bulk directory must be provided."
 436            raise ValueError(msg)
 437        bulk_entry, bulk_locpot = _read_dir(directory_map["bulk"])
 438
 439        def_entries = []
 440        for qq, q_dir in directory_map.items():
 441            if qq == "bulk":
 442                continue
 443            q_entry, q_locpot = _read_dir(q_dir)
 444            q_d_entry = DefectEntry(
 445                defect=defect,
 446                charge_state=int(qq),
 447                sc_entry=q_entry,
 448            )
 449
 450            q_d_entry.get_freysoldt_correction(
 451                defect_locpot=q_locpot,
 452                bulk_locpot=bulk_locpot,
 453                dielectric=dielectric,
 454            )
 455            def_entries.append(q_d_entry)
 456        if vbm is None:
 457            vr = Vasprun(get_zfile(Path(directory_map["bulk"]), "vasprun.xml"))
 458            vbm = vr.get_band_structure().get_vbm()["energy"]
 459
 460        return cls(
 461            bulk_entry=bulk_entry,
 462            defect_entries=def_entries,
 463            pd_entries=pd_entries,
 464            vbm=vbm,
 465            **kwargs,
 466        )

 489    def _vbm_formation_energy(self, defect_entry: DefectEntry, chempots: dict) -> float:
 490        """Compute the formation energy at the VBM.
 491
 492        Compute the formation energy at the VBM (essentially the y-intercept)
 493        for a given defect entry and set of chemical potentials.
 494        If the `bulk_entry` attribute is set, this will be used
 495        difference will be computed using that value.
 496
 497        Args:
 498            defect_entry:
 499                The defect entry for which the formation energy is computed.
 500            chempots:
 501                A dictionary of chemical potentials for each element.
 502
 503        Returns:
 504            float:
 505                The formation energy at the VBM.
 506        """
 507        chempots = self._parse_chempots(chempots)
 508        en_change = sum(
 509            [
 510                (self.dft_energies[Element(el)] + chempots[Element(el)]) * fac
 511                for el, fac in defect_entry.defect.element_changes.items()
 512            ],
 513        )
 514
 515        try:
 516            ediff = defect_entry.get_ediff()
 517        except RuntimeError:
 518            ediff = defect_entry.corrected_energy - self.bulk_entry.energy
 519
 520        return ediff - en_change + self.vbm * defect_entry.charge_state

 585    def get_transitions(
 586        self,
 587        chempots: dict,
 588        x_min: float = 0,
 589        x_max: float | None = None,
 590    ) -> list[tuple[float, float]]:
 591        """Get the transition levels for the formation energy diagram.
 592
 593        Get all of the kinks in the formation energy diagram.
 594        The points at the VBM and CBM are given by the first and last
 595        point respectively.
 596
 597        Args:
 598            chempots:
 599                A dictionary of the chemical potentials (referenced to the elements)
 600                representations a vertex of the stability region of the chemical
 601                potential diagram.
 602            x_min:
 603                The minimum x value (Energy) for the transition levels.
 604            x_max:
 605                The maximum x value (Energy) for the transition levels. If None, the band gap is used.
 606
 607        Returns:
 608            Transition levels and the formation energy at each transition level.
 609            The first and last points are the intercepts with the
 610            VBM and CBM respectively.
 611        """
 612        chempots = self._parse_chempots(chempots)
 613        if x_max is None:  # pragma: no cover
 614            x_max = self.band_gap

 668    def get_chempots(self, rich_element: Element | str, en_tol: float = 0.01) -> dict:
 669        """Get the chemical potential for a desired growth condition.
 670
 671        Choose an element to be rich in, require the chemical potential of that element
 672        to be near the MAX energy among the points (MAX_EN - en_tol, MAX_EN), then sort
 673        the remaining elements by:
 674            1. Are they in the bulk structure:
 675                elements in the bulk structure are prioritized.
 676            2. how similar they in electron affinity to the rich element:
 677                dis-similar elements are prioritized.
 678
 679        .. note::
 680            Priority 2 is pretty arbitrary, but it is a good starting point.
 681
 682
 683        Based on that priority list, take the minimum chemical potential extrema.
 684
 685        Args:
 686            rich_element: The element that are rich in the growth condition.
 687            en_tol: Energy tolerance for the chemical potential of the rich element.
 688
 689        Returns:
 690            A dictionary of the chemical potentials for the growth condition.
 691        """
 692        rich_element = Element(rich_element)
 693        max_val = max(self.chempot_limits, key=lambda x: x[rich_element])[rich_element]
 694        rich_conditions = list(
 695            filter(
 696                lambda cp: abs(cp[rich_element] - max_val) < en_tol,
 697                self.chempot_limits,
 698            ),
 699        )
 700        if len(rich_conditions) == 0:  # pragma: no cover
 701            msg = f"Cannot find a chemical potential condition with {rich_element} near zero."
 702            raise ValueError(
 703                msg,
 704            )
 705        # defect = self.defect_entries[0].defect
 706        in_bulk = self.defect_entries[0].sc_entry.composition.elements
 707        # make sure they are of type Element
 708        in_bulk = [Element(x.symbol) for x in in_bulk]
 709        not_in_bulk = list(set(self.chempot_limits[0].keys()) - set(in_bulk))
 710        in_bulk = list(filter(lambda x: x != rich_element, in_bulk))
 711
 712        def el_sorter(element: Element) -> float:
 713            return -abs(element.electron_affinity - rich_element.electron_affinity)
 714
 715        el_list = sorted(in_bulk, key=el_sorter) + sorted(not_in_bulk, key=el_sorter)
 716
 717        def chempot_sorter(chempot_dict: dict) -> tuple[float, ...]:
 718            return tuple(chempot_dict[el] for el in el_list)
 719
 720        return min(rich_conditions, key=chempot_sorter)

 785    def solve_for_fermi_level(
 786        self,
 787        chempots: dict,
 788        temperature: float,
 789        dos: Dos,
 790    ) -> float:
 791        """Solves for the equilibrium fermi level at a given chempot, temperature, density of states.
 792
 793        Args:
 794            chempots: dictionary of chemical potentials to use
 795            temperature: temperature at which to evaluate.
 796            dos: Density of states object. Must contain a structure attribute. If band_gap attribute
 797                is set, then dos band edges be shifted to match it.
 798
 799        Returns:
 800            Equilibrium fermi level with respect to the valence band edge.
 801        """
 802        fdos = FermiDos(dos, bandgap=self.band_gap)
 803        bulk_factor = self.formation_energy_diagrams[
 804            0
 805        ].defect.structure.composition.get_reduced_formula_and_factor()[1]
 806        fdos_factor = fdos.structure.composition.get_reduced_formula_and_factor()[1]
 807        fdos_multiplicity = fdos_factor / bulk_factor
 808        fdos_cbm, fdos_vbm = fdos.get_cbm_vbm()
 809
 810        def _get_chg(fd: FormationEnergyDiagram, ef: float) -> float:
 811            lines = fd._get_lines(chempots=chempots)
 812            return sum(
 813                fd.defect.multiplicity
 814                * charge
 815                * fermi_dirac(vbm_fe + charge * ef, temperature)
 816                for charge, vbm_fe in lines
 817            )
 818
 819        def _get_total_q(ef: float) -> float:
 820            qd_tot = sum(
 821                _get_chg(fd=fd, ef=ef) for fd in self.formation_energy_diagrams
 822            )
 823            qd_tot += fdos_multiplicity * fdos.get_doping(
 824                fermi_level=ef + fdos_vbm,
 825                temperature=temperature,
 826            )
 827            return qd_tot
 828
 829        return bisect(_get_total_q, -1.0, fdos_cbm - fdos_vbm + 1.0)

 982def get_transitions(
 983    lines: list[tuple[float, float]],
 984    x_min: float,
 985    x_max: float,
 986) -> list[tuple[float, float]]:
 987    """Get the "transition" points in a list of lines.
 988
 989    Given a list of lines represented as (m, b) pairs sorted in order of decreasing m.
 990    A "transition" point is a point where adjacent lines in the list intersect.
 991    i.e. intersection points (x_i, y_i) where line i intersects line i+1
 992
 993    Args:
 994        lines: (m, b) format for each line
 995        x_min: minimum x value
 996        x_max: maximum x value
 997
 998    Returns:
 999        List[List[float]]:
1000            List of intersection points, including the boundary points at
1001            x_min and x_max.
1002    """
1003    # make sure the lines are sorted by decreasing slope
1004    lines = sorted(lines, key=lambda x: x[0], reverse=True)
1005    transitions = [(x_min, lines[0][0] * x_min + lines[0][1])]
1006    for i, (m1, b1) in enumerate(lines[:-1]):
1007        m2, b2 = lines[i + 1]
1008        if m1 == m2:
1009            msg = "The slopes (charge states) of the set of lines should be distinct."
1010            raise ValueError(
1011                msg,
1012            )  # pragma: no cover
1013        nx, ny = ((b2 - b1) / (m1 - m2), (m1 * b2 - m2 * b1) / (m1 - m2))
1014        if nx < x_min:
1015            transitions = [(x_min, m2 * x_min + b2)]
1016        elif nx > x_max:
1017            transitions.append((x_max, m1 * x_max + b1))
1018            break
1019        else:
1020            transitions.append((nx, ny))
1021    else:
1022        transitions.append((x_max, lines[-1][0] * x_max + lines[-1][1]))
1023    return transitions

1026def get_lower_envelope(lines: list[tuple[float, float]]) -> list[tuple[float, float]]:
1027    """Get the lower envelope of the formation energy.
1028
1029    Based on the fact that the lower envelope of the lines is
1030    given by the upper convex hull of the points (m, -b) as shown in:
1031    https://www.cs.umd.edu/class/spring2020/cmsc754/Lects/lect06-duality.pdf
1032    Note: The lines are returned with decreasing slope.
1033
1034    Args:
1035        lines: (m, b) format for each line
1036
1037    Returns:
1038        List[List[float]]:
1039            List lines that make up the lower envelope.
1040    """
1041
1042    def _hash_float(x: float) -> float:
1043        return round(x, 10)
1044
1045    lines_dd: dict = collections.defaultdict(lambda: float("inf"))
1046    for m, b in lines:
1047        lines_dd[_hash_float(m)] = min(lines_dd[_hash_float(m)], b)
1048    lines = list(lines_dd.items())
1049
1050    if len(lines) < 1:  # pragma: no cover
1051        msg = "Need at least one line to get lower envelope."
1052        raise ValueError(msg)
1053    if len(lines) == 1:
1054        return lines
1055    if len(lines) == 2:
1056        return sorted(lines)
1057
1058    dual_points = [(m, -b) for m, b in lines]
1059    upper_hull = get_upper_hull(dual_points)
1060    return [(m, -b) for m, b in upper_hull]

1063def get_upper_hull(points: ArrayLike) -> list[ArrayLike]:
1064    """Get the upper hull of a set of points in 2D.
1065
1066    Args:
1067        points:
1068            List of points in 2D.
1069
1070    Returns:
1071        List[(float, float)]:
1072            Vertices in the upper hull given from right to left.
1073
1074    """
1075    hull = ConvexHull(points)
1076    vertices = hull.vertices
1077
1078    # the vertices are returned in counter-clockwise order
1079    # so we just need to loop over the ring and get the portion
1080    # between the rightmost and leftmost points
1081    right_most_idx = max(vertices, key=lambda x: points[x][0])
1082    left_most_idx = min(vertices, key=lambda x: points[x][0])
1083    seen_right_most = False
1084    upper_hull = []
1085
1086    # loop over the vertices twice
1087    for i in vertices.tolist() + vertices.tolist():
1088        if i == right_most_idx:
1089            seen_right_most = True
1090        if seen_right_most:
1091            xi, yi = points[i]
1092            upper_hull.append((xi, yi))
1093        if seen_right_most and i == left_most_idx:
1094            break
1095    return upper_hull

1098def _get_adjusted_pd_entries(
1099    phase_diagram: PhaseDiagram, atomic_entries: Sequence[ComputedEntry]
1100) -> list[ComputedEntry]:
1101    """Get the adjusted entries for the phase diagram.
1102
1103    Combine the terminal energies from ``atomic_entries`` with the enthalpies of formation
1104    for the provided ``phase_diagram``.  To create the entries for a new phase diagram.

1143def fermi_dirac(energy: float, temperature: float) -> float:
1144    """Get value of fermi dirac distribution.
1145
1146    Gets the defects equilibrium concentration (up to the multiplicity factor)
1147    at a particular fermi level, chemical potential, and temperature (in Kelvin),
1148    assuming dilute limit thermodynamics (non-interacting defects) using FD statistics.
1149
1150    Args:
1151        energy: Energy of the defect with respect to the VBM.
1152        temperature: Temperature in Kelvin.
1153    """
1154    return 1.0 / (1.0 + np.exp((energy) / (boltzman_eV_K * temperature)))

669    def get_chempots(self, rich_element: Element | str, en_tol: float = 0.01) -> dict:
670        """Get the chemical potential for a desired growth condition.
671
672        Choose an element to be rich in, require the chemical potential of that element
673        to be near the MAX energy among the points (MAX_EN - en_tol, MAX_EN), then sort
674        the remaining elements by:
675            1. Are they in the bulk structure:
676                elements in the bulk structure are prioritized.
677            2. how similar they in electron affinity to the rich element:
678                dis-similar elements are prioritized.
679
680        .. note::
681            Priority 2 is pretty arbitrary, but it is a good starting point.
682
683
684        Based on that priority list, take the minimum chemical potential extrema.
685
686        Args:
687            rich_element: The element that are rich in the growth condition.
688            en_tol: Energy tolerance for the chemical potential of the rich element.
689
690        Returns:
691            A dictionary of the chemical potentials for the growth condition.
692        """
693        rich_element = Element(rich_element)
694        max_val = max(self.chempot_limits, key=lambda x: x[rich_element])[rich_element]
695        rich_conditions = list(
696            filter(
697                lambda cp: abs(cp[rich_element] - max_val) < en_tol,
698                self.chempot_limits,
699            ),
700        )
701        if len(rich_conditions) == 0:  # pragma: no cover
702            msg = f"Cannot find a chemical potential condition with {rich_element} near zero."
703            raise ValueError(
704                msg,
705            )
706        # defect = self.defect_entries[0].defect
707        in_bulk = self.defect_entries[0].sc_entry.composition.elements
708        # make sure they are of type Element
709        in_bulk = [Element(x.symbol) for x in in_bulk]
710        not_in_bulk = list(set(self.chempot_limits[0].keys()) - set(in_bulk))
711        in_bulk = list(filter(lambda x: x != rich_element, in_bulk))
712
713        def el_sorter(element: Element) -> float:
714            return -abs(element.electron_affinity - rich_element.electron_affinity)
715
716        el_list = sorted(in_bulk, key=el_sorter) + sorted(not_in_bulk, key=el_sorter)
717
718        def chempot_sorter(chempot_dict: dict) -> tuple[float, ...]:
719            return tuple(chempot_dict[el] for el in el_list)
720
721        return min(rich_conditions, key=chempot_sorter)

670        """Get the chemical potential for a desired growth condition.
671
672        Choose an element to be rich in, require the chemical potential of that element
673        to be near the MAX energy among the points (MAX_EN - en_tol, MAX_EN), then sort
674        the remaining elements by:
675            1. Are they in the bulk structure:
676                elements in the bulk structure are prioritized.
677            2. how similar they in electron affinity to the rich element:
678                dis-similar elements are prioritized.
679
680        .. note::
681            Priority 2 is pretty arbitrary, but it is a good starting point.
682
683
684        Based on that priority list, take the minimum chemical potential extrema.
685
686        Args:
687            rich_element: The element that are rich in the growth condition.
688            en_tol: Energy tolerance for the chemical potential of the rich element.
689
690        Returns:
691            A dictionary of the chemical potentials for the growth condition.
692        """
693        rich_element = Element(rich_element)
694        max_val = max(self.chempot_limits, key=lambda x: x[rich_element])[rich_element]
695        rich_conditions = list(
696            filter(
697                lambda cp: abs(cp[rich_element] - max_val) < en_tol,
698                self.chempot_limits,
699            ),
700        )
701        if len(rich_conditions) == 0:  # pragma: no cover
702            msg = f"Cannot find a chemical potential condition with {rich_element} near zero."
703            raise ValueError(
704                msg,
705            )
706        # defect = self.defect_entries[0].defect
707        in_bulk = self.defect_entries[0].sc_entry.composition.elements
708        # make sure they are of type Element
709        in_bulk = [Element(x.symbol) for x in in_bulk]
710        not_in_bulk = list(set(self.chempot_limits[0].keys()) - set(in_bulk))
711        in_bulk = list(filter(lambda x: x != rich_element, in_bulk))
712
713        def el_sorter(element: Element) -> float:
714            return -abs(element.electron_affinity - rich_element.electron_affinity)
715
716        el_list = sorted(in_bulk, key=el_sorter) + sorted(not_in_bulk, key=el_sorter)
717

719            return tuple(chempot_dict[el] for el in el_list)
720
