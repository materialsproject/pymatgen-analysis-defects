  69class QModel(MSONable):
  70    """Model for the defect charge distribution.
  71
  72    A combination of exponential tail and gaussian distribution is used
  73    (see Freysoldt (2011), DOI: 10.1002/pssb.201046289 )
  74
  75    q_model(r) = q [x exp(-r/gamma) + (1-x) exp(-r^2/beta^2)]
  76    without normalization constants
  77
  78    By default, gaussian distribution with 1 Bohr width is assumed.
  79    If defect charge is more delocalized, exponential tail is suggested.
  80    """
  81
  82    def __init__(
  83        self, beta: float = 1.0, expnorm: float = 0.0, gamma: float = 1.0
  84    ) -> None:
  85        """Initialize the model.
  86
  87        Args:
  88            beta: Gaussian decay constant. Default value is 1 Bohr.
  89                When delocalized (eg. diamond), 2 Bohr is more appropriate.
  90            expnorm: Weight for the exponential tail in the range of [0-1].
  91                Default is 0.0 indicating no tail. For delocalized charges ideal value
  92                is around 0.54-0.6.
  93            gamma: Exponential decay constant
  94        """
  95        self.beta = beta
  96        self.expnorm = expnorm
  97        self.gamma = gamma
  98
  99        self.beta2 = beta * beta
 100        self.gamma2 = gamma * gamma
 101        if expnorm and not gamma:
 102            msg = "Please supply exponential decay constant."
 103            raise ValueError(msg)
 104
 105    def rho_rec(self, g2: float) -> float:
 106        r"""Reciprocal space model charge value.
 107
 108        Reciprocal space model charge value, for input squared reciprocal vector.
 109
 110        Args:
 111            g2: Square of reciprocal vector
 112
 113        Returns:
 114            Charge density at the reciprocal vector magnitude
 115
 116        Note:
 117            Earlier versions used :math:`1/\sqrt{1 + \gamma^2 g^2}` for the
 118            exponential term, which is not the Fourier transform of
 119            :math:`\exp(-r/\gamma)` and disagreed with :attr:`rho_rec_limit0`.
 120            Corrections using an exponential tail (``expnorm > 0``) change as a result,
 121            e.g. by ~4% for ``QModel(expnorm=0.5, gamma=1.5)`` in a 10 Angstrom cubic cell.
 122            The default Gaussian model (``expnorm=0``) is unaffected.
 123        """
 124        # (normalised) Fourier transform of exp(-r/gamma) is 1/(1 + gamma^2 g^2)^2:
 125        return self.expnorm / (1 + self.gamma2 * g2) ** 2 + (1 - self.expnorm) * np.exp(
 126            -0.25 * self.beta2 * g2
 127        )
 128
 129    @property
 130    def rho_rec_limit0(self) -> float:
 131        """Reciprocal space model charge value.
 132
 133        Close to reciprocal vector 0 .
 134        rho_rec(g->0) -> 1 + rho_rec_limit0 * g^2
 135        """
 136        return -2 * self.gamma2 * self.expnorm - 0.25 * self.beta2 * (1 - self.expnorm)

 139def eV_to_k(energy: float) -> float:
 140    """Convert energy to reciprocal vector magnitude k via hbar*k^2/2m.
 141
 142    Args:
 143        energy: Energy in eV.
 144
 145    Returns:
 146        (double) Reciprocal vector magnitude (units of 1/Bohr).
 147    """
 148    return math.sqrt(energy / invang_to_ev) * ang_to_bohr

 151def genrecip(
 152    a1: npt.ArrayLike, a2: npt.ArrayLike, a3: npt.ArrayLike, encut: float
 153) -> Generator[npt.ArrayLike, None, None]:
 154    """Generate reciprocal lattice vectors within the energy cutoff.
 155
 156    Args:
 157        a1: Lattice vector a (in Bohrs)
 158        a2: Lattice vector b (in Bohrs)
 159        a3: Lattice vector c (in Bohrs)
 160        encut: energy cut off in eV
 161
 162    Returns:
 163        reciprocal lattice vectors with energy less than encut
 164    """
 165    vol = np.dot(a1, np.cross(a2, a3))  # 1/bohr^3
 166    b1 = (2 * np.pi / vol) * np.cross(a2, a3)  # units 1/bohr
 167    b2 = (2 * np.pi / vol) * np.cross(a3, a1)
 168    b3 = (2 * np.pi / vol) * np.cross(a1, a2)
 169
 170    # create list of recip space vectors that satisfy |i*b1+j*b2+k*b3|<=encut
 171    G_cut = eV_to_k(encut)
 172    # Figure out max in all recipricol lattice directions
 173    i_max = math.ceil(G_cut / norm(b1))
 174    j_max = math.ceil(G_cut / norm(b2))
 175    k_max = math.ceil(G_cut / norm(b3))
 176
 177    # Build index list
 178    i = np.arange(-i_max, i_max)
 179    j = np.arange(-j_max, j_max)
 180    k = np.arange(-k_max, k_max)
 181
 182    # Convert index to vectors using meshgrid
 183    indices = np.array(np.meshgrid(i, j, k)).T.reshape(-1, 3)
 184    # Multiply integer vectors to get recipricol space vectors
 185    vecs = np.dot(indices, [b1, b2, b3])
 186    # Calculate radii of all vectors
 187    radii = np.sqrt(np.einsum("ij,ij->i", vecs, vecs))
 188
 189    # Yield based on radii
 190    for vec, r in zip(vecs, radii, strict=False):
 191        if r < G_cut and r != 0:
 192            yield vec

 195def converge(f: Callable, step: float, tol: float, max_h: float) -> float:
 196    """Simple newton iteration based convergence function.
 197
 198    Args:
 199        f: function to converge
 200        step: step size for newton iteration
 201        tol: tolerance for convergence
 202        max_h: maximum value of h to try before giving up
 203
 204    Returns:
 205        converged value of f
 206    """
 207    g = f(0)
 208    dx = 10000
 209    h = step
 210    while dx > tol:
 211        g2 = f(h)
 212        dx = abs(g - g2)
 213        g = g2
 214        h += step
 215
 216        if h > max_h:
 217            msg = f"Did not converge before {h}"
 218            raise Exception(msg)
 219    return g

 222def get_zfile(
 223    directory: Path,
 224    base_name: str,
 225    allow_missing: bool = False,
 226) -> Path | None:
 227    """Find gzipped or non-gzipped versions of a file in a directory listing.
 228

 253def generic_group_labels(list_in: Sequence, comp: Callable = operator.eq) -> list[int]:
 254    """Group a list of unsortable objects.
 255
 256    Args:
 257        list_in: A list of generic objects
 258        comp: (Default value = operator.eq) The comparator
 259
 260    Returns:
 261        list[int]: list of labels for the input list
 262
 263    """
 264    list_out: list[int | None] = [None] * len(list_in)
 265    label_num = 0
 266    for i1, ls1 in enumerate(list_out):
 267        if ls1 is not None:
 268            continue
 269        list_out[i1] = label_num
 270        for i2, ls2 in list(enumerate(list_out))[(i1 + 1) :]:
 271            if comp(list_in[i1], list_in[i2]):
 272                if ls2 is None:
 273                    list_out[i2] = list_out[i1]
 274                else:
 275                    list_out[i1] = ls2
 276                    label_num -= 1
 277        label_num += 1
 278    return list_out

 281def get_local_extrema(chgcar: VolumetricData, find_min: bool = True) -> npt.NDArray:
 282    """Get all local extrema fractional coordinates in charge density.
 283
 284    Searching for local minimum by default. Note that sites are NOT grouped
 285    symmetrically.
 286
 287    Args:
 288        chgcar: charge density data
 289        find_min (bool): True to find local minimum else maximum, otherwise
 290            find local maximum.
 291
 292    Returns:
 293        extrema_coords (list): list of fractional coordinates corresponding
 294            to local extrema.
 295    """
 296    sign = -1 if find_min else 1
 297
 298    # Make 3x3x3 supercell
 299    # This is a trick to resolve the periodical boundary issue.
 300    # TODO: Add code to pyrho for max and min filtering.
 301    total_chg = sign * chgcar.data["total"]
 302    total_chg = np.tile(total_chg, reps=(3, 3, 3))
 303    coordinates = peak_local_max(total_chg, min_distance=1)
 304
 305    # Remove duplicated sites introduced by supercell.
 306    f_coords = [coord / total_chg.shape * 3 for coord in coordinates]
 307    f_coords = [
 308        f - 1 for f in f_coords if all(np.array(f) < 2) and all(np.array(f) >= 1)
 309    ]
 310
 311    return np.array(f_coords)

 339def cluster_nodes(
 340    fcoords: npt.ArrayLike,
 341    lattice: Lattice,
 342    tol: float = 0.2,
 343) -> npt.NDArray:
 344    """Cluster nodes that are too close together using hiercharcal clustering.
 345
 346    Args:
 347        fcoords (npt.ArrayLike): fractional coordinates of points to cluster.
 348        lattice (Lattice): The lattice of the structure.
 349        tol (float): A distance tolerance. PBC is taken into account.
 350    """
 351    if len(fcoords) <= 1:  # pragma: no cover
 352        return fcoords
 353    # Manually generate the distance matrix (which needs to take into
 354    # account PBC.
 355    dist_matrix = np.array(lattice.get_all_distances(fcoords, fcoords))
 356    dist_matrix = (dist_matrix + dist_matrix.T) / 2
 357
 358    for i in range(len(dist_matrix)):
 359        dist_matrix[i, i] = 0
 360    condensed_m = squareform(dist_matrix)
 361    z = linkage(condensed_m)
 362    cn = fcluster(z, tol, criterion="distance")
 363    merged_fcoords = []
 364
 365    for n in set(cn):
 366        frac_coords = []
 367        for i, j in enumerate(np.where(cn == n)[0]):
 368            if i == 0:
 369                frac_coords.append(fcoords[j])
 370            else:
 371                fcoord = fcoords[j]
 372                # We need the image to combine the frac_coords properly.
 373                _d, image = lattice.get_distance_and_image(frac_coords[0], fcoord)
 374                frac_coords.append(fcoord + image)
 375        merged_fcoords.append(np.average(frac_coords, axis=0))
 376
 377    merged_fcoords = [f - np.floor(f) for f in merged_fcoords]
 378    merged_fcoords = [f * (np.abs(f - 1) > 1e-15) for f in merged_fcoords]
 379    # the second line for fringe cases like
 380    # np.array([ 5.0000000e-01 -4.4408921e-17  5.0000000e-01])
 381    # where the shift to [0,1) does not work due to float precision
 382
 383    return np.array(merged_fcoords)

 386def get_avg_chg(
 387    chgcar: VolumetricData,
 388    fcoord: npt.ArrayLike,
 389    radius: float = 0.4,
 390) -> float:
 391    """Get the average charge in a sphere.
 392
 393    Args:
 394        chgcar: The charge density.
 395        fcoord: The fractional coordinates of the center of the sphere.
 396        radius: The radius of the sphere in Angstroms.
 397
 398    Returns:
 399        The average charge in the sphere.
 400
 401
 402    """
 403    # makesure fcoord is an array
 404    fcoord = np.array(fcoord)
 405
 406    def _dist_mat(pos_frac: npt.ArrayLike) -> npt.NDArray:
 407        # return a matrix that contains the distances
 408        aa = np.linspace(0, 1, len(chgcar.get_axis_grid(0)), endpoint=False)
 409        bb = np.linspace(0, 1, len(chgcar.get_axis_grid(1)), endpoint=False)
 410        cc = np.linspace(0, 1, len(chgcar.get_axis_grid(2)), endpoint=False)
 411        AA, BB, CC = np.meshgrid(aa, bb, cc, indexing="ij")
 412        dist_from_pos = chgcar.structure.lattice.get_all_distances(
 413            np.vstack([AA.flatten(), BB.flatten(), CC.flatten()]).T,
 414            pos_frac,
 415        )
 416        return dist_from_pos.reshape(AA.shape)
 417
 418    if np.any(fcoord < 0) or np.any(fcoord > 1):
 419        msg = "f_coords must be in [0,1)"
 420        raise ValueError(msg)
 421    mask = _dist_mat(fcoord) < radius
 422    vol_sphere = chgcar.structure.volume * (mask.sum() / chgcar.ngridpts)
 423    return np.sum(chgcar.data["total"] * mask) / mask.size / vol_sphere

 426class TopographyAnalyzer:
 427    """Topography Analyzer.
 428
 429    This is a generalized module to perform topological analyses of a crystal
 430    structure using Voronoi tessellations. It can be used for finding potential
 431    interstitial sites. Applications including using these sites for
 432    inserting additional atoms or for analyzing diffusion pathways.
 433
 434    Note that you typically want to do some preliminary postprocessing after
 435    the initial construction. The initial construction will create a lot of
 436    points, especially for determining potential insertion sites. Some helper
 437    methods are available to perform aggregation and elimination of nodes. A
 438    typical use is something like::
 439
 440        a = TopographyAnalyzer(structure, ["O", "P"])
 441    """
 442
 443    def __init__(
 444        self,
 445        structure: Structure,
 446        framework_ions: list[str],
 447        cations: list[str],
 448        image_tol: float = 0.0001,
 449        max_cell_range: int = 1,
 450        check_volume: bool = True,
 451        constrained_c_frac: float = 0.5,
 452        thickness: float = 0.5,
 453        clustering_tol: float = 0.5,
 454        min_dist: float = 0.9,
 455        ltol: float = 0.2,
 456        stol: float = 0.3,
 457        angle_tol: float = 5,
 458    ) -> None:
 459        """Initialize the TopographyAnalyzer.
 460
 461        Args:
 462            structure (Structure): An initial structure.
 463            framework_ions ([str]): A list of ions to be considered as a
 464                framework. Typically, this would be all anion species. E.g.,
 465                ["O", "S"].
 466            cations ([str]): A list of ions to be considered as non-migrating
 467                cations. E.g., if you are looking at Li3PS4 as a Li
 468                conductor, Li is a mobile species. Your cations should be [
 469                "P"]. The cations are used to exclude polyhedra from
 470                diffusion analysis since those polyhedra are already occupied.
 471            image_tol (float): A tolerance distance for the analysis, used to
 472                determine if something are actually periodic boundary images of
 473                each other. Default is usually fine.
 474            max_cell_range (int): This is the range of periodic images to
 475                construct the Voronoi tessellation. A value of 1 means that we
 476                include all points from (x +- 1, y +- 1, z+- 1) in the
 477                voronoi construction. This is because the Voronoi poly
 478                extends beyond the standard unit cell because of PBC.
 479                Typically, the default value of 1 works fine for most
 480                structures and is fast. But for very small unit
 481                cells with high symmetry, this may need to be increased to 2
 482                or higher. If there are < 5 atoms in the input structure and
 483                max_cell_range is 1, this will automatically be increased to 2.
 484            check_volume (bool): Set False when ValueError always happen after
 485                tuning tolerance.
 486            constrained_c_frac (float): Constraint the region where users want
 487                to do Topology analysis the default value is 0.5, which is the
 488                fractional coordinate of the cell
 489            thickness (float): Along with constrained_c_frac, limit the
 490                thickness of the regions where we want to explore. Default is
 491                0.5, which is mapping all the site of the unit cell.
 492            clustering_tol (float): Tolerance for clustering nodes. Default is
 493                0.5.
 494            min_dist (float): Minimum distance between nodes. Default is 0.9.
 495            ltol (float): Lattice parameter tolerance for structure matching.
 496                Default is 0.2.
 497            stol (float): Site tolerance for structure matching. Default is
 498                0.3.
 499            angle_tol (float): Angle tolerance for structure matching. Default
 500                is 5.
 501
 502        """
 503        self.framework_ions = {get_el_sp(sp) for sp in framework_ions}
 504        self.cations = {get_el_sp(sp) for sp in cations}
 505        self.clustering_tol = clustering_tol
 506        self.min_dist = min_dist
 507        self.sm = StructureMatcher(ltol=ltol, stol=stol, angle_tol=angle_tol)
 508
 509        # if input cell is very small (< 5 atoms) and max cell range is 1 (default), bump to 2 for
 510        # accurate Voronoi tessellation:
 511        if len(structure) < 5 and max_cell_range == 1:
 512            max_cell_range = 2
 513
 514        # Let us first map all sites to the standard unit cell, i.e.,
 515
 516        # 0 ≤ coordinates < 1.
 517        # structure = Structure.from_sites(structure, to_unit_cell=True)
 518        # lattice = structure.lattice
 519
 520        # We could constrain the region where we want to dope/explore by setting
 521        # the value of constrained_c_frac and thickness. The default mode is
 522        # mapping all sites to the standard unit cell
 523
 524        self.structure = structure.copy()
 525        # NOTE: Structure is still being mutated something weird is going on but the code works.
 526        # remove oxidation state
 527        self.structure.remove_oxidation_states()
 528
 529        constrained_sites = []
 530        for _i, site in enumerate(self.structure):
 531            if (
 532                site.frac_coords[2] >= constrained_c_frac - thickness
 533                and site.frac_coords[2] <= constrained_c_frac + thickness
 534            ):
 535                constrained_sites.append(site)
 536        constrained_struct = Structure.from_sites(sites=constrained_sites)
 537        lattice = constrained_struct.lattice
 538
 539        # Divide the sites into framework and non-framework sites.
 540        framework = []
 541        non_framework = []
 542        for site in constrained_struct:
 543            # site.species can be Composition SpecieLike
 544            if hasattr(site.species, "elements"):
 545                # Handle the case where site.species is a Composition
 546                els = [*map(Element, site.species.elements)]
 547            else:
 548                els = [*map(get_el_sp, site.species.keys())]
 549            if self.framework_ions.intersection(els):
 550                framework.append(site)
 551            else:
 552                non_framework.append(site)
 553
 554        # We construct a supercell series of coords. This is because the
 555        # Voronoi polyhedra can extend beyond the standard unit cell. Using a
 556        # range of -2, -1, 0, 1 should be fine.
 557        coords = []
 558        cell_range = list(range(-max_cell_range, max_cell_range + 1))
 559        for shift in itertools.product(cell_range, cell_range, cell_range):
 560            for site in framework:
 561                shifted = site.frac_coords + shift
 562                coords.append(lattice.get_cartesian_coords(shifted))
 563
 564        # Perform the voronoi tessellation.
 565        voro = Voronoi(coords)
 566
 567        # Store a mapping of each voronoi node to a set of points.
 568        node_points_map = defaultdict(set)
 569        for pts, vs in voro.ridge_dict.items():
 570            for v in vs:
 571                node_points_map[v].update(pts)
 572
 573        _logger.debug("Voronoi vertices in cell: %s", len(voro.vertices))
 574
 575        # Vnodes store all the valid voronoi polyhedra. Cation vnodes store
 576        # the voronoi polyhedra that are already occupied by existing cations.
 577        vnodes: list[VoronoiPolyhedron] = []
 578        cation_vnodes = []
 579
 580        def get_mapping(poly: VoronoiPolyhedron) -> VoronoiPolyhedron | None:
 581            """Helper function.
 582
 583            Checks if a vornoi poly is a periodic image of
 584            one of the existing voronoi polys.
 585            """
 586            for v in vnodes:
 587                if v.is_image(poly, image_tol):
 588                    return v
 589            return None
 590
 591        # Filter all the voronoi polyhedra so that we only consider those
 592        # which are within the unit cell.
 593        for i, vertex in enumerate(voro.vertices):
 594            if i == 0:
 595                continue
 596            fcoord = lattice.get_fractional_coords(vertex)
 597            poly = VoronoiPolyhedron(lattice, fcoord, node_points_map[i], coords, i)
 598            if np.all([-image_tol <= c < 1 + image_tol for c in fcoord]):
 599                if len(vnodes) == 0:
 600                    vnodes.append(poly)
 601                else:
 602                    ref = get_mapping(poly)
 603                    if ref is None:
 604                        vnodes.append(poly)
 605
 606        _logger.debug("%s - voronoi vertices in cell.", len(vnodes))
 607
 608        # Eliminate all voronoi nodes which are closest to existing cations.
 609        if len(cations) > 0:
 610            cation_coords = [
 611                site.frac_coords
 612                for site in non_framework
 613                if self.cations.intersection(site.species.keys())
 614            ]
 615
 616            vertex_fcoords = [v.frac_coords for v in vnodes]
 617            dist_matrix = lattice.get_all_distances(cation_coords, vertex_fcoords)
 618            indices = np.where(dist_matrix == np.min(dist_matrix, axis=1)[:, None])[1]
 619            cation_vnodes = [v for i, v in enumerate(vnodes) if i in indices]
 620            vnodes = [v for i, v in enumerate(vnodes) if i not in indices]
 621
 622        _logger.debug("%s - vertices in cell not with cation.", len(vnodes))
 623        self.coords = coords
 624        self.vnodes = vnodes
 625        self.cation_vnodes = cation_vnodes
 626        self.framework = framework
 627        self.non_framework = non_framework
 628        if check_volume:
 629            self.check_volume()
 630
 631    @property
 632    def labeled_sites(
 633        self,
 634    ) -> list[tuple[list[float], int]]:
 635        """Get a list of inserted structures and a list of structure matching labels.
 636
 637        Returns:
 638            list[tuple[list[float], int]]: A list of tuples of the form (fcoords, label)
 639        """
 640        # Get a reasonablly reduced set of candidate sites first
 641        candidate_sites = [v.frac_coords for v in self.vnodes]
 642        return get_labeled_inserted_structure(
 643            candidate_sites,

 649        )

 652        """Basic check for volume of all voronoi poly sum to unit cell volume.
 653
 654        Note that this does not apply after poly combination.
 655        """
 656        vol = sum(v.volume for v in self.vnodes) + sum(
 657            v.volume for v in self.cation_vnodes
 658        )
 659        if abs(vol - self.structure.volume) > 1e-8:  # pragma: no cover

 662                "structure! This may lead to inaccurate results. You need to "
 663                "tweak the tolerance and max_cell_range until you get a "
 664                "correct mapping."

 681class VoronoiPolyhedron:
 682    """Convenience container for a voronoi point in PBC and its associated polyhedron."""
 683
 684    def __init__(
 685        self,
 686        lattice: Lattice,
 687        frac_coords: npt.ArrayLike,
 688        polyhedron_indices: list | set,
 689        all_coords: list,
 690        name: str | int | None = None,
 691    ) -> None:
 692        """Initialize a VoronoiPolyhedron.
 693
 694        Args:
 695            lattice (Lattice): Lattice of the structure.
 696            frac_coords (np.array): Fractional coordinates of the voronoi point.
 697            polyhedron_indices (list): Indices of the polyhedron vertices.
 698            all_coords (list): List of all polyhedron vertices.
 699            name (str): Name of the polyhedron.
 700        """
 701        self.lattice = lattice
 702        self.frac_coords = frac_coords
 703        self.polyhedron_indices = polyhedron_indices
 704        self.polyhedron_coords = np.array(all_coords)[list(polyhedron_indices), :]
 705        self.name = name
 706
 707    def is_image(self, poly: VoronoiPolyhedron, tol: float) -> bool:
 708        """Check if poly is an image of the current polyhedron.
 709
 710        Args:
 711            poly (VoronoiPolyhedron): Polyhedron to check.
 712            tol (float): Tolerance for image check.
 713
 714        Returns:
 715            bool: True if poly is an image of the current polyhedron.
 716        """
 717        frac_diff = pbc_diff(poly.frac_coords, self.frac_coords)
 718        if not np.allclose(frac_diff, [0, 0, 0], atol=tol):
 719            return False
 720        to_frac = self.lattice.get_fractional_coords
 721        for c1 in self.polyhedron_coords:
 722            found = False
 723            for c2 in poly.polyhedron_coords:
 724                d = pbc_diff(to_frac(c1), to_frac(c2))
 725                if not np.allclose(d, [0, 0, 0], atol=tol):
 726                    found = True
 727                    break
 728            if not found:
 729                return False
 730        return True

 733    def coordination(self) -> int:
 734        """Coordination number."""
 735        return len(self.polyhedron_indices)
 736
 737    @property
 738    def volume(self) -> float:
 739        """Volume of the polyhedron."""
 740        return calculate_vol(self.polyhedron_coords)
 741
 742    def __str__(self) -> str:
 743        """String representation."""
 744        return f"Voronoi polyhedron {self.name}"

 855def _get_ipr(spin: int, k_index: int, procar: Procar) -> npt.NDArray:
 856    states = procar.data[spin][k_index, ...]
 857    flat_states = states.reshape(states.shape[0], -1)
 858    return 1 / np.sum(flat_states**2, axis=1)

 861def get_ipr_in_window(
 862    bandstructure: BandStructure,
 863    procar: Procar,
 864    band_window: int = 5,
 865) -> dict[Spin, npt.NDArray]:
 866    """Get the inverse participation ratio (IPR) of the states in a given band window.
 867
 868    For a given number of bands above and below the Fermi level, the IPR is calculated.
 869
 870    Args:
 871        bandstructure: The bandstructure object.
 872            The band just below the fermi level is used as the center of band window.
 873        procar: The procar object.
 874        band_window: The number of bands above and blow the fermi level to include in the search window.
 875
 876    Returns:
 877        dict[(int, int), npt.NDArray]: The IPR of the states in the band window keyed for each k-point and spin.
 878    """
 879    res = {}
 880    for spin in bandstructure.bands:
 881        s_index = 0 if spin == Spin.up else 1
 882        # last band that is fully below the fermi level
 883        last_occ_idx = bisect.bisect_left(
 884            bandstructure.bands[spin].max(1),
 885            bandstructure.efermi,
 886        )
 887        lbound = max(last_occ_idx - band_window, 0)
 888        ubound = min(last_occ_idx + band_window, bandstructure.nb_bands)
 889        for k_idx, _ in enumerate(bandstructure.kpoints):
 890            ipr = _get_ipr(spin, k_idx, procar)
 891            res[(k_idx, s_index)] = np.stack(
 892                (np.arange(lbound, ubound), ipr[lbound:ubound]),
 893            ).T
 894    return res

 897def get_localized_states(
 898    bandstructure: BandStructure,
 899    procar: Procar,
 900    band_window: int = 7,
 901) -> Generator[tuple[int, int, int, float], None, None]:
 902    """Find the (band, kpt, spin) index of the most localized state.
 903
 904    Find the band index with the lowest inverse participation ratio (IPR) in a small window
 905    around the fermi level.  Since some of the core states can be very localized,
 906    we should only search near the band edges.  Report the result for each k-point and spin channel.
 907
 908    Args:
 909        bandstructure: The bandstructure object.
 910            The band just below the fermi level is used as the center of band window.
 911        procar: The procar object.
 912        band_window: The number of bands above and blow the fermi level to include in the search window.
 913
 914    Returns:
 915        tuple(int, int, int): Returns the (band, kpt, spin) index of the most localized state.
 916    """
 917    d_ib_ipr = get_ipr_in_window(bandstructure, procar, band_window)
 918    for (k_index, ispin), ib_ipr in d_ib_ipr.items():
 919        # find the index of the minimum IPR
 920        min_idx, val = min(ib_ipr, key=lambda x: x[1])
 921        yield (int(min_idx), k_index, ispin, val)

 924def sort_positive_definite(
 925    list_in: list,
 926    ref1: object,
 927    ref2: object,
 928    dist: Callable,
 929) -> tuple[tuple, tuple[float]]:
 930    """Sort a list where we can only compute a positive-definite distance.
 931
 932    Sometimes, we can only compute a positive-definite distance between two objects.
 933    (E.g., the displacement between two structures). In these cases, standard
 934    sorting algorithms will not work. Here, we accept two reference points to give
 935    some sense of direction. We then sort the list based on the distance between the
 936    reference points. Note: this only works if the list falls on a line of some sort.
 937
 938    Args:
 939        list_in: The list to sort.
 940        ref1: The first reference point, this will be the `zero` point.
 941        ref2: The second reference point, this will determine the direction.
 942        dist: Some positive definite distance function.
 943
 944    Returns:
 945        - the sorted list of objects.
 946        - the signed distance in the chosen direction.
 947    """
 948    d1 = [dist(el, ref1) for el in list_in]
 949    d2 = [dist(el, ref2) for el in list_in]
 950    D0 = dist(ref1, ref2)
 951
 952    d_vs_s = []
 953    for q1, q2, el in zip(d1, d2, list_in, strict=False):
 954        sign = +1
 955        if q1 < q2 and q2 > D0:
 956            sign = -1
 957        d_vs_s.append((sign * q1, el))
 958    d_vs_s.sort()
 959    distances, sorted_list = list(zip(*d_vs_s, strict=False))
 960    return sorted_list, distances

 980def get_labeled_inserted_structure(
 981    sites: npt.NDArray,
 982    host_structure: Structure,
 983    working_ion: str,
 984    min_dist: float,
 985    clustering_tol: float,
 986    sm: StructureMatcher,
 987) -> list[tuple[list[float], int]]:
 988    """Get a list of inserted structures and a list of structure matching labels.
 989
 990    The process is as follows:
 991    1. Read in the list of candidate sites.
 992    2. Remove any sites that collide with the host and cluster the remaining sites.
 993    2. Group the candidate sites by symmetry by inserting the working ion and structure matching.
 994    3. Label the groups by structure matching.
 995
 996    Args:
 997        sites: The candidate sites to insert.
 998        host_structure: The host structure.
 999        working_ion: The working ion.
1000        min_dist: The minimum distance between the working ion and atoms in the host structure.
1001        clustering_tol: The tolerance for clustering the candidate sites.
1002        sm: The StructureMatcher object to use for grouping the sites.
1003
1004    Returns:
1005        list[tuple[list[float], int]]: A list of tuples of the form (fcoords, label)
1006    """
1007    sites = remove_collisions(sites, structure=host_structure, min_dist=min_dist)
1008    sites = cluster_nodes(sites, lattice=host_structure.lattice, tol=clustering_tol)
1009
1010    # Group the candidate sites by symmetry
1011    inserted_structs = []
1012    for fpos in sites:
1013        tmp_struct = host_structure.copy()
1014        get_valid_magmom_struct(tmp_struct, inplace=True, spin_mode="none")
1015        tmp_struct.insert(
1016            0,
1017            working_ion,
1018            fpos,
1019        )
1020        tmp_struct.sort()
1021        inserted_structs.append(tmp_struct)
1022
1023    # Label the groups by structure matching
1024    site_labels = generic_group_labels(inserted_structs, comp=sm.fit)
1025    return [*zip(sites.tolist(), site_labels, strict=False)]

1064    docs: Sequence,
1065    sm: StructureMatcher,
1066    get_structure: Callable,
1067    get_hash: Callable | None = None,
1068) -> Generator[tuple[str | None, list], None, None]:
1069    """Group docs by a simple hash followed by structure.
1070

1083        Generator of (name, group)

1091            sgroups = list(_group_docs_by_structure(list(g), sm, get_structure))
1092            if len(sgroups) <= 1:
1093                for group in sgroups:
1094                    yield h, group
1095            else:
1096                for itr, group in enumerate(sgroups):
1097                    yield f"{h}:{itr}", group

1100def get_plane_spacing(lattice: npt.NDArray) -> list[float]:
1101    """Compute the plane spacing of a lattice.
1102
1103    Get the cartesian spacing between periodic planes of a unit cell.
1104    Copied from materialsproject/pyrho.
1105
1106    >>> get_plane_spacing([[1,0,0], [1,1,0], [0,0,2]])
1107    [0.7653..., 1.042..., 2.0]
1108
1109    Args:
1110    lattice:
1111        List of lattice vectors in cartesian coordinates
1112
1113    Returns:
1114    List[float]:
1115        List where the k-th element is is the spacing of planes generated by all
1116        lattice vectors EXCEPT the k-th one
1117
1118    """
1119    # get all pairwise projections i must be smaller than j
1120    ndim = len(lattice)
1121    idx_pairs = [*itertools.combinations(range(ndim), 2)]
1122    latt_len = [np.linalg.norm(lattice[i]) for i in range(ndim)]
1123    pproj = {
1124        (i, j): np.dot(lattice[i], lattice[j]) / latt_len[i] / latt_len[j]
1125        for i, j in idx_pairs
1126    }
1127    # get the spacing in each direction:
1128    spacing = []
1129    for idir in range(ndim):
1130        idir_proj = [
1131            np.array(lattice[j]) * pproj[tuple(sorted([idir, j]))]  # type: ignore[index]
1132            for j in range(ndim)
1133            if j != idir
1134        ]
1135        v_perp_subspace = lattice[idir] - sum(idir_proj)
1136        spacing.append(np.linalg.norm(v_perp_subspace))
1137    return spacing

154) -> Generator[npt.ArrayLike, None, None]:
155    """Generate reciprocal lattice vectors within the energy cutoff.
156
157    Args:
158        a1: Lattice vector a (in Bohrs)
159        a2: Lattice vector b (in Bohrs)
160        a3: Lattice vector c (in Bohrs)
161        encut: energy cut off in eV
162
163    Returns:
164        reciprocal lattice vectors with energy less than encut
165    """
166    vol = np.dot(a1, np.cross(a2, a3))  # 1/bohr^3
167    b1 = (2 * np.pi / vol) * np.cross(a2, a3)  # units 1/bohr
168    b2 = (2 * np.pi / vol) * np.cross(a3, a1)
169    b3 = (2 * np.pi / vol) * np.cross(a1, a2)
170
171    # create list of recip space vectors that satisfy |i*b1+j*b2+k*b3|<=encut
172    G_cut = eV_to_k(encut)
173    # Figure out max in all recipricol lattice directions
174    i_max = math.ceil(G_cut / norm(b1))
175    j_max = math.ceil(G_cut / norm(b2))
176    k_max = math.ceil(G_cut / norm(b3))
177
178    # Build index list
179    i = np.arange(-i_max, i_max)
180    j = np.arange(-j_max, j_max)
181    k = np.arange(-k_max, k_max)
182
183    # Convert index to vectors using meshgrid
184    indices = np.array(np.meshgrid(i, j, k)).T.reshape(-1, 3)
185    # Multiply integer vectors to get recipricol space vectors
186    vecs = np.dot(indices, [b1, b2, b3])
187    # Calculate radii of all vectors
188    radii = np.sqrt(np.einsum("ij,ij->i", vecs, vecs))
189
190    # Yield based on radii
191    for vec, r in zip(vecs, radii, strict=False):
192        if r < G_cut and r != 0:
193            yield vec
